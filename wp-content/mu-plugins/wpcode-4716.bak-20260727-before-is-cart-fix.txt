// บล็อกการใช้แต้มเกินแบบกันทุกกรณี (click/submit/Enter) และทนต่อ AJAX ที่รีเฟรช DOM
add_action('wp_footer', function () {
  if (!(is_cart() || is_checkout())) return; ?>
<script>
jQuery(function($){

  /* ====== Config: ตัวกรองแบบกว้าง ไม่ผูกปลั๊กอินใด ====== */
  const FIELD_SEL = 'input[type="number"], input[type="text"]';
  const BTN_SEL   = 'button, input[type="submit"], input[type="button"], a.button';

  function isPointsField(el){
    const s = (el.id+' '+el.name+' '+el.className).toLowerCase();
    return /point|redeem|wpr/.test(s);
  }
  function isApplyPointsBtn(el){
    const t = ($(el).text()+' '+($(el).val()||'')+' '+el.className+' '+el.id).toLowerCase();
    return /apply.*point|point.*apply|ใช้แต้ม|แลกแต้ม|redeem/.test(t);
  }

  // หา field แต้มตัวแรกที่น่าจะใช้งาน
  function getPointsField(){
    const $candidates = $(FIELD_SEL).filter(function(){ return isPointsField(this); });
    return $candidates.first();
  }

  // อ่านแต้มที่มีอยู่ (รองรับไทย/อังกฤษ)
  function getAvailable($field){
    // 1) ถ้ามี max ให้ใช้ก่อน
    const m = parseInt($field.attr('max'),10);
    if (!isNaN(m) && m>0) return m;

    // 2) หา text รอบ ๆ
    const $wrap = $field.closest('.wps_cart_points_wrapper, .mwb_wpr_points_wrapper, .woocommerce, form');
    const txt = $wrap.text();
    let mm = txt.match(/available\s*points[^0-9]*([0-9,]+)/i) ||
             txt.match(/แต้ม(?:ที่มี|คงเหลือ)?[^0-9]*([0-9,]+)/i);
    if (mm) return parseInt(mm[1].replace(/,/g,''),10);

    // 3) สำรอง: มองหาเลขใกล้ ๆ
    const near = ($field.nextAll().text()||'')+' '+($field.prevAll().text()||'');
    mm = near.match(/([0-9][0-9,]{0,6})/);
    return mm ? parseInt(mm[1].replace(/,/g,''),10) : NaN;
  }

  function getRequested($field){
    return parseInt(String($field.val()||'').replace(/[^0-9]/g,''),10) || 0;
  }

  function showInlineError($field, msg){
    $field.nextAll('.wps-points-error-inline').remove();
    $('<div class="wps-points-error-inline" style="color:#b81c23;font-size:13px;margin-top:6px;">'+msg+'</div>')
      .insertAfter($field);
    $field.addClass('woocommerce-invalid');
    setTimeout(()=>{$field.removeClass('woocommerce-invalid');},1500);
  }

  function validatePoints(){
    const $field = getPointsField();
    if (!$field.length) return true; // ไม่เจอ field ก็ปล่อยผ่าน
    const avail = getAvailable($field);
    const want  = getRequested($field);
    if (!isNaN(avail) && want > avail){
      showInlineError($field, 'แต้มไม่พอสำหรับการแลกสิทธิ์');
      // ตั้ง max กันกรอกเกินในครั้งต่อไป
      $field.attr('max', avail);
      return false;
    }
    return true;
  }

  /* ====== ดักทุกกรณี ====== */

  // 1) คลิกปุ่ม Apply Points (ใช้ delegated handler ทนต่อ DOM ถูกแทนที่)
  $(document).on('click', BTN_SEL, function(e){
    if (!isApplyPointsBtn(this)) return;
    if (!validatePoints()){
      e.preventDefault(); e.stopImmediatePropagation(); return false;
    }
  });

  // 2) Submit ฟอร์ม (เช่น ผู้ใช้กด Enter)
  $(document).on('submit', 'form', function(e){
    // ตรวจว่า submit นี้เกิดจากปุ่ม Apply Points หรือช่องแต้มโฟกัสอยู่
    const sub = (e.originalEvent && e.originalEvent.submitter) || document.activeElement || null;
    if (sub && isApplyPointsBtn(sub)){
      if (!validatePoints()){ e.preventDefault(); e.stopImmediatePropagation(); return false; }
    } else {
      // กรณี Enter บนช่องแต้มโดยไม่มีปุ่ม submitter
      const $field = getPointsField();
      if ($field.length && $field.is(':focus') && !validatePoints()){
        e.preventDefault(); e.stopImmediatePropagation(); return false;
      }
    }
  });

  // 3) กันผู้ใช้พิมพ์เกินแบบ real-time
  $(document).on('input', FIELD_SEL, function(){
    if (!isPointsField(this)) return;
    const $f = $(this);
    const avail = getAvailable($f);
    const want  = getRequested($f);
    if (!isNaN(avail) && want > avail){ $f.val(avail); }
  });

});
</script>
<?php });
